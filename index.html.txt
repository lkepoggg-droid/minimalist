    function createLanterns() {
      var box = document.getElementById('lanternLayer');
      var positions = [15, 32, 55, 72, 86];
      for (var i = 0; i < positions.length; i++) {
        var l = document.createElement('div');
        l.className = 'sky-lantern';
        l.style.left = positions[i] + '%';
        l.style.animationDuration = (9 + i * 2.2) + 's';
        l.style.animationDelay = (i * 1.8) + 's';
        box.appendChild(l);
      }
    }

      try { localStorage.setItem('krathong-pond-v1', JSON.stringify(state)); } catch (e) {}
      renderAll();
    }

    function renderAll() {
      var c = state.config;
      document.getElementById('uiMainTitle').textContent = c.mainTitle;
      document.getElementById('uiSubTitle').textContent = c.subTitle;
      document.getElementById('uiHqName').textContent = c.hqName;
      document.getElementById('uiWarnSign').textContent = c.warnSign;
      document.getElementById('pondAnimalIcon').textContent = c.animal || '🐊';
      document.getElementById('croc1').textContent = c.animal || '🐊';

      var chipsBox = document.getElementById('tagChips');
      chipsBox.innerHTML = '';
      (c.tags || []).forEach(function (t) {
        var btn = document.createElement('button');
        btn.type = 'button';
        btn.className = 'chip';
        btn.textContent = t;
        btn.addEventListener('click', function () {
          document.getElementById('inPlayerTag').value = t;
        });
        chipsBox.appendChild(btn);
      });

      var layer = document.getElementById('krathongLayer');
      layer.innerHTML = '';
      state.krathongs.forEach(function (item) {
        var node = document.createElement('div');
        node.className = 'krathong-node';
        node.style.left = item.left + '%';
        node.style.bottom = item.bottom + 'px';
        node.style.animationDelay = item.delay || '0s';

        var emojiDiv = document.createElement('div');
        emojiDiv.className = 'k-emoji';
        emojiDiv.textContent = item.icon;

        var labelDiv = document.createElement('div');
        labelDiv.className = 'k-label';

        node.appendChild(labelDiv);

          showToast('💬 ' + item.name + ': "' + (item.wish || 'สาธุ ขอให้ปังๆ!') + '"');
        });

      });

      if (id === 'settingsModal') {
        var c = state.config;
        document.getElementById('cfgMainTitle').value = c.mainTitle;
        document.getElementById('cfgSubTitle').value = c.subTitle;
        document.getElementById('cfgHqName').value = c.hqName;
        document.getElementById('cfgWarnSign').value = c.warnSign;
        document.getElementById('cfgAnimal').value = c.animal;
        document.getElementById('cfgTags').value = (c.tags || []).join(', ');
      }
      document.getElementById(id).classList.add('open');
    }

    function closeModal(id) {
      document.getElementById(id).classList.remove('open');
    }

    function launchFirework() {
      var colors = ['#facc15', '#38bdf8', '#f472b6', '#4ade80', '#c084fc', '#fb923c'];
      for (var burst = 0; burst < 2; burst++) {
        var cx = window.innerWidth * (0.25 + Math.random() * 0.5);
        var cy = window.innerHeight * (0.16 + Math.random() * 0.24);
        for (var i = 0; i < 22; i++) {
          var p = document.createElement('div');
          p.className = 'firework-particle';
          var angle = (Math.PI * 2 * i) / 22;
          var dist = 45 + Math.random() * 65;
          p.style.left = cx + 'px';
          p.style.top = cy + 'px';
          p.style.background = colors[(i + burst) % colors.length];
          p.style.setProperty('--dx', Math.cos(angle) * dist + 'px');
          p.style.setProperty('--dy', Math.sin(angle) * dist + 'px');
          document.body.appendChild(p);
          (function (el) {
            setTimeout(function () { el.remove(); }, 1000);
          })(p);
        }
      }
    }

    var toastTimer = null;
    function showToast(msg) {
      var old = document.querySelector('.toast');
      if (old) old.remove();
      clearTimeout(toastTimer);
      var el = document.createElement('div');
      el.className = 'toast';
      el.textContent = msg;
      document.body.appendChild(el);
      toastTimer = setTimeout(function () { el.remove(); }, 2200);
    }
